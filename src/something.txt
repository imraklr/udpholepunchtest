use std::io::{self, Write};
use std::net::{SocketAddr, UdpSocket};
use std::thread;
use std::time::{Duration, Instant};

const STUN_SERVER: &str = "stun.l.google.com:19302";
const STUN_BINDING_REQUEST: u16 = 0x0001;

// STUN magic cookie defined by RFC 5389.
const MAGIC_COOKIE: u32 = 0x2112_A442;

fn main() -> io::Result<()> {
    println!("=== UDP Hole Punching Test ===");
    println!();

    // ---------------------------------------------------------
    // 1. Create UDP socket
    // ---------------------------------------------------------

    let socket = UdpSocket::bind("0.0.0.0:0")?;

    let local_addr = socket.local_addr()?;

    println!("Local UDP socket: {local_addr}");
    println!();

    // ---------------------------------------------------------
    // 2. Discover our public UDP endpoint using STUN
    // ---------------------------------------------------------

    println!("Querying STUN server: {STUN_SERVER}");

    let public_addr = stun_binding_request(&socket)?;

    println!();
    println!("Your public UDP endpoint:");
    println!("    {public_addr}");
    println!();

    println!("Give this address to the other laptop.");
    println!();

    // ---------------------------------------------------------
    // 3. Ask for the other peer's public endpoint
    // ---------------------------------------------------------

    let peer_addr = loop {
        print!("Enter the other laptop's public UDP endpoint: ");
        io::stdout().flush()?;

        let mut input = String::new();
        io::stdin().read_line(&mut input)?;

        match input.trim().parse::<SocketAddr>() {
            Ok(addr) => break addr,
            Err(_) => {
                println!("Invalid address. Example: 203.0.113.10:42137");
            }
        }
    };

    println!();
    println!("Peer endpoint: {peer_addr}");
    println!();

    // ---------------------------------------------------------
    // 4. Start hole punching
    // ---------------------------------------------------------

    println!("Starting UDP hole punching...");
    println!("Both laptops should be running this program.");
    println!();

    socket.set_read_timeout(Some(Duration::from_millis(100)))?;

    let receiver = socket.try_clone()?;

    // Receiver thread.
    let receive_thread = thread::spawn(move || {
        let mut buf = [0u8; 2048];

        let start = Instant::now();

        loop {
            // Stop after 30 seconds.
            if start.elapsed() > Duration::from_secs(30) {
                break;
            }

            match receiver.recv_from(&mut buf) {
                Ok((len, source)) => {
                    let message = String::from_utf8_lossy(&buf[..len]);

                    println!();
                    println!("<<< RECEIVED FROM {source}");
                    println!("<<< {message}");
                    println!();

                    if message.starts_with("HOLE_PUNCH") {
                        println!("======================================");
                        println!("SUCCESS: UDP hole punch established!");
                        println!("======================================");
                        println!();
                    }
                }

                Err(e)
                    if e.kind() == io::ErrorKind::WouldBlock
                        || e.kind() == io::ErrorKind::TimedOut =>
                {
                    // Nothing received yet.
                }

                Err(e) => {
                    eprintln!("Receive error: {e}");
                    break;
                }
            }
        }
    });

    // ---------------------------------------------------------
    // 5. Send packets repeatedly
    // ---------------------------------------------------------

    for i in 1..=60 {
        let message = format!(
            "HOLE_PUNCH packet={} from={}",
            i, local_addr
        );

        match socket.send_to(message.as_bytes(), peer_addr) {
            Ok(bytes) => {
                println!(
                    ">>> Sent {} bytes to {} [packet {}]",
                    bytes, peer_addr, i
                );
            }

            Err(e) => {
                eprintln!("Send error: {e}");
            }
        }

        thread::sleep(Duration::from_millis(500));
    }

    println!();
    println!("Finished sending hole-punch packets.");
    println!("Waiting for remaining packets...");

    receive_thread.join().ok();

    println!();
    println!("Test finished.");

    Ok(())
}

// =============================================================
// STUN
// =============================================================

fn stun_binding_request(socket: &UdpSocket) -> io::Result<SocketAddr> {
    let server: SocketAddr = STUN_SERVER
        .to_socket_addrs()?
        .next()
        .ok_or_else(|| {
            io::Error::new(
                io::ErrorKind::Other,
                "Could not resolve STUN server",
            )
        })?;

    // ---------------------------------------------------------
    // Construct STUN Binding Request
    //
    // STUN header:
    //
    // Bytes 0-1   : Message Type
    // Bytes 2-3   : Message Length
    // Bytes 4-7   : Magic Cookie
    // Bytes 8-19  : Transaction ID
    // ---------------------------------------------------------

    let mut packet = [0u8; 20];

    // Message Type = Binding Request (0x0001)
    packet[0..2].copy_from_slice(&STUN_BINDING_REQUEST.to_be_bytes());

    // Message Length = 0
    packet[2..4].copy_from_slice(&0u16.to_be_bytes());

    // Magic Cookie
    packet[4..8].copy_from_slice(&MAGIC_COOKIE.to_be_bytes());

    // Transaction ID
    //
    // We don't need cryptographic randomness for this simple
    // experiment. Generate a value based on the current time
    // and local socket information.
    let transaction_id = generate_transaction_id(socket);

    packet[8..20].copy_from_slice(&transaction_id);

    // Send STUN request.
    socket.send_to(&packet, server)?;

    println!("STUN request sent to {server}");

    // ---------------------------------------------------------
    // Wait for response
    // ---------------------------------------------------------

    socket.set_read_timeout(Some(Duration::from_secs(5)))?;

    let mut response = [0u8; 2048];

    let (len, source) = socket.recv_from(&mut response)?;

    println!("STUN response received from {source}");

    // Restore normal timeout later.
    socket.set_read_timeout(None)?;

    parse_stun_response(&response[..len], &transaction_id)
}

// =============================================================
// STUN response parser
// =============================================================

fn parse_stun_response(
    packet: &[u8],
    transaction_id: &[u8; 12],
) -> io::Result<SocketAddr> {
    if packet.len() < 20 {
        return Err(io::Error::new(
            io::ErrorKind::InvalidData,
            "STUN response is too short",
        ));
    }

    // Message type.
    let message_type = u16::from_be_bytes([packet[0], packet[1]]);

    // Binding Success Response = 0x0101
    if message_type != 0x0101 {
        return Err(io::Error::new(
            io::ErrorKind::InvalidData,
            format!(
                "Unexpected STUN message type: 0x{:04x}",
                message_type
            ),
        ));
    }

    // Verify magic cookie.
    let cookie =
        u32::from_be_bytes([packet[4], packet[5], packet[6], packet[7]]);

    if cookie != MAGIC_COOKIE {
        return Err(io::Error::new(
            io::ErrorKind::InvalidData,
            "Invalid STUN magic cookie",
        ));
    }

    // Verify transaction ID.
    if &packet[8..20] != transaction_id {
        return Err(io::Error::new(
            io::ErrorKind::InvalidData,
            "STUN transaction ID does not match",
        ));
    }

    let message_length =
        u16::from_be_bytes([packet[2], packet[3]]) as usize;

    let end = 20 + message_length;

    if packet.len() < end {
        return Err(io::Error::new(
            io::ErrorKind::InvalidData,
            "Truncated STUN response",
        ));
    }

    // ---------------------------------------------------------
    // Parse STUN attributes.
    // ---------------------------------------------------------

    let mut pos = 20;

    while pos + 4 <= end {
        let attr_type =
            u16::from_be_bytes([packet[pos], packet[pos + 1]]);

        let attr_len =
            u16::from_be_bytes([packet[pos + 2], packet[pos + 3]])
                as usize;

        pos += 4;

        if pos + attr_len > end {
            return Err(io::Error::new(
                io::ErrorKind::InvalidData,
                "Invalid STUN attribute length",
            ));
        }

        let value = &packet[pos..pos + attr_len];

        match attr_type {
            // XOR-MAPPED-ADDRESS
            0x0020 => {
                if let Some(addr) =
                    parse_xor_mapped_address(value, transaction_id)
                {
                    return Ok(addr);
                }
            }

            // MAPPED-ADDRESS (older STUN)
            0x0001 => {
                if let Some(addr) = parse_mapped_address(value) {
                    return Ok(addr);
                }
            }

            _ => {}
        }

        // STUN attributes are padded to a 4-byte boundary.
        pos += (attr_len + 3) & !3;
    }

    Err(io::Error::new(
        io::ErrorKind::NotFound,
        "No XOR-MAPPED-ADDRESS or MAPPED-ADDRESS in STUN response",
    ))
}

// =============================================================
// XOR-MAPPED-ADDRESS
// =============================================================

fn parse_xor_mapped_address(
    value: &[u8],
    transaction_id: &[u8; 12],
) -> Option<SocketAddr> {
    if value.len() < 8 {
        return None;
    }

    let family = value[1];

    // IPv4
    if family == 0x01 {
        let port =
            u16::from_be_bytes([value[2], value[3]])
                ^ ((MAGIC_COOKIE >> 16) as u16);

        let cookie = MAGIC_COOKIE.to_be_bytes();

        let ip = [
            value[4] ^ cookie[0],
            value[5] ^ cookie[1],
            value[6] ^ cookie[2],
            value[7] ^ cookie[3],
        ];

        return Some(SocketAddr::from((ip, port)));
    }

    // IPv6
    if family == 0x02 && value.len() >= 20 {
        let port =
            u16::from_be_bytes([value[2], value[3]])
                ^ ((MAGIC_COOKIE >> 16) as u16);

        let cookie = MAGIC_COOKIE.to_be_bytes();

        let mut mask = [0u8; 16];

        mask[..4].copy_from_slice(&cookie);
        mask[4..].copy_from_slice(transaction_id);

        let mut ip = [0u8; 16];

        for i in 0..16 {
            ip[i] = value[4 + i] ^ mask[i];
        }

        return Some(SocketAddr::from((ip, port)));
    }

    None
}

// =============================================================
// MAPPED-ADDRESS
// =============================================================

fn parse_mapped_address(value: &[u8]) -> Option<SocketAddr> {
    if value.len() < 8 {
        return None;
    }

    let family = value[1];

    if family == 0x01 {
        let port = u16::from_be_bytes([value[2], value[3]]);

        let ip = [value[4], value[5], value[6], value[7]];

        return Some(SocketAddr::from((ip, port)));
    }

    None
}

// =============================================================
// Transaction ID generation
// =============================================================

fn generate_transaction_id(socket: &UdpSocket) -> [u8; 12] {
    use std::time::{SystemTime, UNIX_EPOCH};

    let now = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default();

    let nanos = now.as_nanos();

    let local_port = socket
        .local_addr()
        .map(|a| a.port() as u128)
        .unwrap_or(0);

    let mut value = nanos ^ (local_port << 32);

    let mut result = [0u8; 12];

    for byte in result.iter_mut() {
        value ^= value << 13;
        value ^= value >> 7;
        value ^= value << 17;

        *byte = value as u8;
        value >>= 1;
    }

    result
}
